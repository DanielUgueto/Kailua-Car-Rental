USE kailua_car_rental;

-- tables
CREATE TABLE IF NOT EXISTS category
(
    categoryID     INT AUTO_INCREMENT PRIMARY KEY,
    type           VARCHAR(100)                          NOT NULL UNIQUE,
    min_horsepower INT                                   NOT NULL DEFAULT 0,
    min_seats      INT                                   NOT NULL DEFAULT 1,
    min_cc         INT                                   NOT NULL DEFAULT 0,
    gear           ENUM ('Automatic', 'Manual', 'Other') NOT NULL


);

CREATE TABLE IF NOT EXISTS cars
(
    carID          INT AUTO_INCREMENT PRIMARY KEY,
    categoryID     INT                                                                  NOT NULL,
    FOREIGN KEY (categoryID) REFERENCES category (categoryID),
    brand          VARCHAR(100)                                                         NOT NULL,
    model          VARCHAR(100)                                                         NOT NULL,
    fuel_type      ENUM ('Gasoline', 'Petrol', 'Diesel', 'Hybrid', 'Electric', 'Other') NOT NULL,
    reg_number     VARCHAR(50)                                                          NOT NULL UNIQUE,
    reg_date       DATE                                                                 NOT NULL,
    odometerKM     INT                                                                  NOT NULL DEFAULT 0,
    cruise_control BOOLEAN                                                              NOT NULL DEFAULT FALSE,
    leather_seats  BOOLEAN                                                              NULL     DEFAULT FALSE,
    engine_cc      INT                                                                  NOT NULL,
    automatic_gear BOOLEAN                                                              NOT NULL DEFAULT FALSE,
    air_condition  BOOLEAN                                                              NOT NULL DEFAULT TRUE,
    seats          TINYINT                                                              NOT NULL DEFAULT 1,
    horsepower     INT                                                                  NOT NULL DEFAULT 0
);

CREATE TABLE IF NOT EXISTS customer
(
    customerID            INT AUTO_INCREMENT PRIMARY KEY,
    first_name            VARCHAR(50)  NOT NULL,
    last_name             VARCHAR(50)  NOT NULL,
    address               VARCHAR(255) NOT NULL,
    zip_code              VARCHAR(50)  NOT NULL,
    city                  VARCHAR(50)  NOT NULL,
    personal_phone        VARCHAR(50)  NOT NULL UNIQUE,
    work_phone            VARCHAR(50)  NUll,
    email                 VARCHAR(50)  NULL,
    country               VARCHAR(50)  NOT NULL DEFAULT 'Denmark',
    driver_license_number VARCHAR(50)  NOT NULL UNIQUE,
    driver_since          DATE         NOT NULL
);

CREATE TABLE IF NOT EXISTS car_rental
(
    rentalID       INT AUTO_INCREMENT PRIMARY KEY,
    categoryID     INT      NOT NULL,
    FOREIGN KEY (categoryID) REFERENCES category (categoryID),
    carID          INT      NOT NULL,
    FOREIGN KEY (carID) REFERENCES cars (carID),
    customerID     INT      NOT NULL,
    FOREIGN KEY (customerID) REFERENCES customer (customerID),
    rental_start   DATETIME NOT NULL,
    rental_end     DATETIME NOT NULL,
    minKM          INT      NOT NULL DEFAULT 10,
    maxKM          INT      NOT NULL DEFAULT 50,
    odometer_start INT      NOT NULL,
    odometer_end   INT      NOT NULL
);
